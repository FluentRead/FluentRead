const marker = 'INJECTED_FOCUS_OBSERVER_FAILURE_DURING_STOP';

module.exports = {
  ...realHelper,
  startFocusEventMonitor(options) {
    const monitor = realHelper.startFocusEventMonitor(options);
    let stopPromise, injectedError;
    const stop = () => {
      if (!stopPromise) stopPromise = monitor.stop().then(value => {
        injectedError = new Error(marker);
        return value;
      });
      return stopPromise;
    };
    return new Proxy(monitor, {
      get(target, property, receiver) {
        if (property === 'stop') return stop;
        if (property === 'error') return target.error || injectedError;
        return Reflect.get(target, property, receiver);
      },
    });
  },
};
